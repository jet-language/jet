// The helper's `answer` collides with the hand-written one below.
facts: fn() -> Code {
    #code {
        Fact: struct { value: Int }
        answer: fn() -> Fact { Fact{value: 42} }
    }
}

build: fn(b: Build) {
    b.generate("facts", facts())
    app: b.add_executable("app", ["main.jet", ".jet/generated/main/facts.jet"], [])
    b.default(app)
}

run: fn() { print("{answer().value}") }

answer: fn() -> Fact { Fact{value: 1} }
