// main.jet with a changed `first`, then a loop that runs until the build's
// step limit, or until the case stops it, before `second` is generated.
build: fn(b: Build) {
    b.write_generated(.On)
    b.generate("first") {
        first_value: fn() -> Int { 41 }
    }
    total := 0
    loop n in 0..4000000000 {
        total = total + n
    }
    b.generate("second") {
        second_value: fn() -> Int { first_value() + 1 }
    }
    app: b.add_executable("app", ["main.jet", ".jet/generated/main/first.jet", ".jet/generated/main/second.jet"], [])
    b.default(app)
}

run: fn() { print("{second_value()}") }
