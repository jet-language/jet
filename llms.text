# Jet LLM surface digest

Generated. Current compiler registries own markers, diagnostics, syntax names, and Core items.
Regenerate with `node scripts/agent/generate-llm-digest.mjs`; CI compares the bytes.
Use active rows only. Retired rows teach replacement; they are not valid current source.
Write one current program. Do not invent aliases, legacy spellings, or library namespaces.

## First program

A source file ends with one `fn run()` entry. `print` is a built-in.

```jet
fn run() {
    greeting :: "Hello, Jet"
    print(greeting)
}
```

No semicolons. Comments start with `//`. Strings use double quotes and interpolate `{name}`.

## Core source rules

Bindings: `name :: value` is immutable; `name := value` is mutable; `name = value` reassigns a mutable binding.
Functions: `fn name(parameter: Type) => Return { ... }`; expression bodies use `:: expression`.
Visibility: declarations are private by default; prefix an item with `pub` for package use.
Types: `Int`, `Float`, `Bool`, `String`, `Char`; lists use `[T]`; optional values use `T?`; failures use `T ? E`.
Errors: handle `T?` or `T ? E` with `?? fallback`, `?`, or a pattern test. Use `Ok(value)`, `Err(error)`, `Val(value)`, and `None`.
Control: `if condition { ... } else { ... }`; collecting loops use `loop name, source { ... }`; exit with `break` and advance with `next`.
Construction: use `Type.{ field: value }`; list literals use `[T].{ value1, value2 }`.
Calls and member access use `name(args)` and `value.member(args)`. Core imports use `use core.module as alias`.
Ownership is safe by default. `&T` writes, `^T` moves, and `~value` copies. Expert unsafe code needs `#Unsafe("reason")`.

## Canonical compiling example

Read this as working source syntax. It is the checked executable showcase in `examples/canon.jet`.

```jet
// canon.jet — the compiling syntax showcase.
//
// Every line here is RATIFIED and IMPLEMENTED Jet syntax that compiles and
// runs through `jet`. Golden-tested in tests/release_gates.rs (was
// tests/canon.rs): if it stops compiling or its output drifts, the suite
// goes red. Treat it as the canonical "what works today" reference.
//
// Aspirational / ratified-but-unimplemented syntax lives in
//   docs/reference/syntax-surface.jet  (non-executable).
// Decision IDs + rationale live in docs/spec/syntax-decisions.md.
use core.mem
use core.encoding.json as json

// --- variables & types ---------------------------------------------------
// `name :: val` / `name := val` / `name = val` bind; types ride values (`Type.{ … }`) or live on signatures/fields (D-BIND-BARE1).
fn show_values() {
    name :: "Jet"
    year :: 2026
    version :: 0.1
    ready :: true
    print("values: {name} {year} {version} {ready}")
    count := 0
    count += 3
    print("mutable: {count}")
}

// --- functions & generics ------------------------------------------------
fn greet(who: String) => String :: "hello, {who}"

fn id<T>(x: ^T) => T :: x

struct Point {
    x: Float
    y: Float

    fn dist_sq(self) => Float :: self.x * self.x + self.y * self.y

    fn origin() => Point {
        return Point.{x: 0.0, y: 0.0}
    }
}

// --- enums & pattern matching --------------------------------------------
// `if subject == { .arm -> body }`; `||` joins arms; `else ->` is default (D-ENUMDOT1).
enum Light {
    Red
    Yellow
    Green
}

fn next(light: Light) => Light {
    if light == {
        .Red -> { return Light.Yellow }
        .Yellow -> { return Light.Green }
        .Green -> { return Light.Red }
    }
}

fn day_kind(day: String) => String {
    if day == {
        "sat" | "sun" -> { return "weekend" }
        else -> { return "weekday" }
    }
}

// --- errors: `T ? E`, ok/err, `?` propagate, `??` fallback ---------------
enum ParseError {
    Empty
}

fn parse_age(raw: String) => Int ? ParseError {
    if raw == "" { return Err(ParseError.Empty) }
    return Ok(7)
}

fn double_age(raw: String) => Int ? ParseError {
    n :: parse_age(raw)?
    return Ok(n * 2)
}

// --- traits & impl -------------------------------------------------------
trait Shape {
    fn area(self) => Float
}

struct Square {
    side: Float
}

impl Square.Shape {
    fn area(self) => Float :: self.side * self.side
}

// --- inline module -------------------------------------------------------
module math {
    pub fn double(n: Int) => Int :: n * 2
}

// --- low-level: `Type.{ uninit }` sentinel --------------------------------
// Opts a binding out of automatic zero-fill; sema proves every read happens
// after a write on all paths (E0420, D-UNINIT-SENTINEL2). Gated by `use core.mem`.
fn show_uninit() => Int {
    n := Int.{ uninit }
    n = 99
    return n
}

// --- inline contracts: `#Inline` / `#Inline(Always)` ----------------------
// D-INLINE-PARAM1=A: `#Inline` is a soft hint; `#Inline(Always)` is a checked
// promise the compiler proves or refuses to build.
#Inline fn square(x: Int) => Int { return x * x }

#Inline(Always) fn twice(x: Int) => Int { return x * 2 }

// --- typed CLI args: `#CLI` struct + `#Doc(...)` fields -----------------
// D-CLIFLAG1=A: an entry function's `#CLI` parameter type IS the CLI spec.
// Constructed by hand here (not run as the CLI entry point) to keep this
// file's single `fn run()` the one program entry.
#CLI
struct ServeArgs {
    #Doc("port to listen on") port: Int = 3000
    #Doc("print extra detail") verbose: Bool
}

// --- stream parsing: `Reader.over` / `Cursor.over` / `take_pattern` -------
// D-SHIFT1: the "shift" kernel for linear stream parsing — every read
// advances position and returns `? Err`, never panics or truncates silently.
fn parse_packet() => String {
    packet :: [U8].{42, 0, 0, 0, 3, 0, 16, 32, 48}
    r :: Reader.over(packet)
    magic :: r.read_u32_le() ?? panic("bad packet: no magic")
    count :: r.read_u16_le() ?? panic("bad packet: no count")
    payload :: r.take(count) ?? panic("bad packet: short payload")
    return "magic={magic} count={count} payload_len={payload.len()}"
}

fn parse_incident(line: String) => String {
    c :: Cursor.over(line)
    c.skip_ws()
    matched :: c.take_pattern("inc-{id:Int} sev {sev:Int}: ") ?? panic("bad incident line")
    reason :: c.take_until("\n") ?? panic("no newline found")
    return "id={matched.id} sev={matched.sev} reason={reason}"
}

// --- decode-time migration transparency: `decode_traced` ------------------
// D-MIGRATE3=A: `decode_traced<T>` returns `{ value, migration }` so a
// caller can ask whether a decoded record arrived as an old schema version.
#[PublishedSchema, Codable]
struct Account {
    id: Int
    display_name: String
    verified: Bool
}

migration Account {
    rename name => display_name
    add verified: Bool = false
}

fn decode_account(raw: String) => String {
    r :: json.decode_traced<Account>(raw) ?? panic("bad account")
    return "id={r.value.id} migrated={r.migration.migrated}"
}

// --- OS-target gating: `#Target(OS.Linux)` on impl ------------------------
// D-OSTARGET1=A: only the impl matching the build's OS is compiled in.
// There is no ratified way yet for ungated code to dispatch to "whichever
// backend matches this build", so — matching
// examples/features/lowlevel/os_target_gating.jet — main() never calls
// this gated impl directly; it only proves the two compile side by side.
trait Backend {
    fn label(self) => String
}

struct LinuxBackend {
    name: String
}

#Target(OS.Linux)
impl LinuxBackend.Backend {
    fn label(self) => String :: "linux: {self.name}"
}

// --- `#Test` scope members: `.setup` / `.expect_fail` / `.timeout` / `.skip`
// D-DOTSCOPE1: contextual dot-member blocks inside a `#Test` marker scope.
// Run by `jet test`, not by this file's `fn run()`.
#Test("uninit then write is provably safe") {
    .setup {
        base :: show_uninit()
    }
    require_eq(base, 99)
}

#Test("expect_fail region must fail") {
    .expect_fail {
        require((1 == 2), "one is not two")
    }
    require_eq(twice(2), 4)
}

#Test("timeout bounds a region") {
    .timeout(500ms) {
        require_eq(square(5), 25)
    }
}

#Test("skip one region") {
    require_eq(twice(3), 6)
    .skip {
        require(false)
    }
}

// --- main: exercise each feature with deterministic output ---------------
fn run() {
    show_values()
    print(greet("world"))
    same :: id(99)
    print("id: {same}")
    p :: Point.{x: 3.0, y: 4.0}
    print("dist_sq: {p.dist_sq()}")
    o :: Point.origin()
    print("origin: {o.x} {o.y}")
    print("next: {next(Light.Red) == Light.Yellow}")
    kind :: day_kind("sun")
    print("day: {kind}")
    // errors
    if double_age("42") == {
        .Ok(n) -> {
            print("ok: {n}")
        }
        .Err(e) -> {
            print("err: {e}")
        }
    }

    fallback :: parse_age("") ?? 0
    print("fallback: {fallback}")
    // collections + closures
    nums := [Int].{3, 1, 2}
    nums.push(4)
    nums.sort()
    squares :: nums.map((n: Int) => n * n)
    big :: squares.filter((n: Int) => n > 5)
    total :: nums.reduce(0, (acc: Int, n: Int) => acc + n)
    print("nums: {nums[0]} {nums[1]} total {total} big {big.len()}")
    // tuple
    pair :: (x: 1, y: 2)
    print("tuple: {pair.x} {pair.y}")
    // trait dispatch
    sq :: Square.{side: 5.0}
    print("area: {sq.area()}")
    // module
    print("module: {math.double(21)}")
    // strings
    parts :: "a,b,c".split(",")
    trimmed :: "  hi  ".trim()
    print("split: {parts.len()} trimmed: {trimmed}")
    // control flow
    sum := 0
    loop i, 1..5 { sum += i }
    print("loop_sum: {sum}")
    // low-level: uninit sentinel
    print("uninit: {show_uninit()}")
    // inline contracts
    print("inline: {square(4)} {twice(5)}")
    // typed CLI args struct (constructed by hand, not run as the entry point)
    args :: ServeArgs.{port: 8080, verbose: true}
    print("cli: {args.port} {args.verbose}")
    // stream parsing
    print("reader: {parse_packet()}")
    incident_line :: "  inc-4411 sev 3: disk full\n"
    print("cursor: {parse_incident(incident_line)}")
    // decode-time migration transparency
    account_json :: "{{\"id\": 7, \"display_name\": \"Betty\", \"verified\": true}}"
    print("decode: {decode_account(account_json)}")
}
```
