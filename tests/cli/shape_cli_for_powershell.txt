# PowerShell completion for build/typed (JetCommandSchema v1)
Register-ArgumentCompleter -Native -CommandName 'build/typed' -ScriptBlock { param($wordToComplete,$commandAst,$cursorPosition) @('--help','--name','--retries','--verbose') | Where-Object { $_ -like "$wordToComplete*" } | ForEach-Object { [System.Management.Automation.CompletionResult]::new($_,$_,'ParameterName',$_) } }
