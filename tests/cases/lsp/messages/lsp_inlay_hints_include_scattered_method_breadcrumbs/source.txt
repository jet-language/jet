DrawThing: trait {
    render: fn(self) -> String
}

Widget: struct {
    title: String
}

impl Widget {
    size: fn(self) -> Int {
        return 1
    }
}

impl Widget.DrawThing {
    render: fn(self) -> String {
        return self.title
    }
}

run: fn() {
    w: Widget{title: "ok"}
    print(w.render())
}
