trait DrawThing {
    fn render(self) -> String
}

struct Widget {
    title: String
}

impl Widget {
    fn size(self) -> Int {
        return 1
    }
}

impl Widget.DrawThing {
    fn render(self) -> String {
        return self.title
    }
}

fn run() {
    w :: Widget{title: "ok"}
    print(w.render())
}
