// A private part of core.text that only this tree's
// Compiler/Prelude/CoreSourceParts.tsv declares.
use {
    segment_tables: __core_text_segmentation_tables
}

tree_part_width: fn() -> Int {
    segment_tables.UNICODE_ALPHABETIC.len()
}

#Test("a part only the tree declares loads with its owner") {
    assert(tree_part_width() > 0)
}
