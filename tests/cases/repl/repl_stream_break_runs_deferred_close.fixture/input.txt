
Marker: struct {
    name: String
}

impl Marker.Close {
    close: fn(^self) {
        print("closed: {self.name}")
    }
}

values: fn() -[]> Stream<Int> {
    marker := Marker{name: "stream"}
    defer close(^marker)
    print("before")
    yield 1
    print("after")
    yield 2
}

run: fn() {
    loop value in values() {
        print("value: {value}")
        break
    }
}

:run
