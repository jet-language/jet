jet dev — Watch and run a program; optionally open Canvas or a local app

  jet dev [<file.jet|dir>] [--canvas|--app <function>] [--share <loopback|lan>] [--token <token>] [-- <args>]
  --gc-trace                   Record bounded automatic GC-promotion evidence
  --json                       Emit machine-readable facts or diagnostics
  --quiet                      Suppress non-error status output
  --color                      Color: auto | always | never
  --output                     Select a named output (build --lib selects a Library; Canvas uses it as the initial selection)
  -p                           Pick a workspace member by name
  --restart                    Always rerun from scratch after a save
  --swap                       Hot-swap compatible edits and restart after type changes
  --watch                      Re-run on dependency changes; test default off with r=failed, a=all, o=output; jobs accepts --watch[=<on|off>]; --watch=off runs once
  --canvas                     Open the full Canvas IDE over this dev session
  --app                        Select a checked local-app function
  --share                      Select local-app sharing authority: loopback or lan
  --canvas-host                With --canvas, bind host (loopback by default)
  --canvas-port                With --canvas, bind an explicit port
  --canvas-transport           With --canvas, select the named transport
  --canvas-authority           With --canvas, select loopback or remote authority
  --canvas-audit               With --canvas, enable request/rebuild audit output
  --token                      Require a bearer token for the session
  --offline                    Reuse only cached signed toolchain and package data
  --fixtures                   Use an explicit fixture bundle
  --gate                       Allow one audited gate as --gate name=allow
  --target                     Select a target (triple, board.<name>, or named database target)
  --env                        Select one declared env.<name> module
  --preset                     Select one declared environment preset
  --no-capture                 Disable automatic replay capture for this session
  --release                    Optimize for release
  --profile                    How hard to optimize: release, debug, ci, hardened, or a named optimization bundle
  --optimize                   How hard to optimize: none, basic or full
  --bounds                     What a failed bounds check does: fatal, report or off
  --debug-info                 Whether the image carries DWARF debug info: true, false or split
  --small                      Favor a smaller binary by keeping inlining to the O1 limits: true or false
  --inline                     Which calls the optimizer inlines: auto, always_marked or off
  --backtrace                  What a fault report shows of the stack: full, brief or off
  --dead-code                  Whether the image keeps the package's unreached functions: remove or keep
  --checked-alloc              Build with the checked memory allocator, which stops at a bad free or a write after free: true or false
  --write-generated            Whether fn build's generated sources are written to .jet/generated/: off or on
  --output-dir                 The directory jet build writes its executables to ("" is .jet/build)
  --libs                       More C libraries the native link resolves foreign symbols against
  --link-args                  More native link inputs: -L<dir>, -l<name> or a .a archive path
  --os                         The operating system the image runs on: linux, macos, windows, web or freestanding
  --min-os                     The oldest OS version the image runs on ("" is none)
  --set                        Set one declared package setting as key=value
  --threads                    Use at most N compiler threads (1 = one thread); without it jet picks a safe number of compiler threads
  --show-default               Use and report the stock default
  --record                     Record a named replay as --record=<name>
